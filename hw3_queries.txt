Dataset/API: Finnhub Stock Market Data
Database: finnhub_db
Collection: companies
Source: Finnhub API (https://finnhub.io)

Each document contains: company profile (name, symbol, country, industry,
exchange, ipo_date, market_cap), nested financial metrics (52WeekHigh,
beta, peRatio, etc.), peer companies list, quarterly earnings history,
and daily price history (OHLCV) for the past 30 days.



QUERY 1: Find all Technology companies, showing name, symbol, and market cap
Question: Which companies in our dataset are in the Technology industry?

db.companies.find(
  { industry: "Technology" },
  { _id: 0, name: 1, symbol: 1, market_cap: 1 }
).sort({ market_cap: -1 })


QUERY 2: Top 10 companies by market capitalization
Question: What are the 10 largest companies by market cap?

db.companies.find(
  { market_cap: { $gt: 0 } },
  { _id: 0, name: 1, symbol: 1, market_cap: 1, industry: 1 }
).sort({ market_cap: -1 }).limit(10)


QUERY 3: Companies with a PE ratio under 15 (value stocks)
Question: Which companies have a trailing PE ratio below 15, suggesting they may be undervalued?

db.companies.find(
  { "metrics.peNormalizedAnnual": { $lt: 15, $gt: 0 } },
  { _id: 0, name: 1, symbol: 1, "metrics.peNormalizedAnnual": 1, industry: 1 }
).sort({ "metrics.peNormalizedAnnual": 1 })


QUERY 4: Average market cap by industry (aggregation)
Question: What is the average market capitalization for each industry?


db.companies.aggregate([
  { $match: { market_cap: { $gt: 0 } } },
  { $group: {
      _id: "$industry",
      avg_market_cap: { $avg: "$market_cap" },
      count: { $sum: 1 }
  }},
  { $sort: { avg_market_cap: -1 } },
  { $project: {
      _id: 0,
      industry: "$_id",
      avg_market_cap: { $round: ["$avg_market_cap", 2] },
      count: 1
  }}
])

QUERY 5: Companies with high beta (volatile stocks)
Question: Which companies have a beta greater than 1.5, indicating high volatility?


db.companies.find(
  { "metrics.beta": { $gt: 1.5 } },
  { _id: 0, name: 1, symbol: 1, "metrics.beta": 1, industry: 1 }
).sort({ "metrics.beta": -1 })


QUERY 6: Count of companies per industry (aggregation)
Question: How many companies do we have in each industry sector?


db.companies.aggregate([
  { $group: {
      _id: "$industry",
      count: { $sum: 1 }
  }},
  { $sort: { count: -1 } },
  { $project: {
      _id: 0,
      industry: "$_id",
      count: 1
  }}
])


QUERY 7: Companies that beat earnings estimates (nested array query)
Question: Which companies beat their most recent quarterly earnings estimate?


db.companies.aggregate([
  { $match: { earnings: { $ne: [] } } },
  { $project: {
      _id: 0, name: 1, symbol: 1,
      latest_actual: { $arrayElemAt: ["$earnings.actual", 0] },
      latest_estimate: { $arrayElemAt: ["$earnings.estimate", 0] },
      latest_period: { $arrayElemAt: ["$earnings.period", 0] }
  }},
  { $match: { $expr: { $gt: ["$latest_actual", "$latest_estimate"] } } },
  { $limit: 10 }
])


QUERY 8: Highest volume trading day per company (nested array - price_history)
Question: For each company, what was the highest volume trading day in the past month?


db.companies.aggregate([
  { $match: { "price_history": { $ne: [] } } },
  { $project: {
      _id: 0, name: 1, symbol: 1,
      max_volume_day: {
        $arrayElemAt: [
          { $filter: {
              input: "$price_history",
              as: "day",
              cond: { $eq: ["$$day.volume", { $max: "$price_history.volume" }] }
          }}, 0
        ]
      }
  }},
  { $sort: { "max_volume_day.volume": -1 } },
  { $limit: 10 }
])


QUERY 9: Monthly return for each stock (price_history first vs last close)
Question: What is the approximate monthly return for each stock based on price history?


db.companies.aggregate([
  { $match: { "price_history": { $ne: [] } } },
  { $project: {
      _id: 0, name: 1, symbol: 1, industry: 1,
      first_close: { $arrayElemAt: ["$price_history.close", 0] },
      last_close: { $arrayElemAt: ["$price_history.close", -1] }
  }},
  { $project: {
      name: 1, symbol: 1, industry: 1,
      first_close: 1, last_close: 1,
      monthly_return_pct: {
        $round: [
          { $multiply: [
            { $divide: [{ $subtract: ["$last_close", "$first_close"] }, "$first_close"] },
            100
          ]}, 2
        ]
      }
  }},
  { $sort: { monthly_return_pct: -1 } },
  { $limit: 10 }
])


QUERY 10: Average daily trading volume by industry (aggregation on nested array)
Question: Which industries have the highest average daily trading volume?


db.companies.aggregate([
  { $match: { "price_history": { $ne: [] } } },
  { $project: {
      industry: 1,
      avg_volume: { $avg: "$price_history.volume" }
  }},
  { $group: {
      _id: "$industry",
      avg_daily_volume: { $avg: "$avg_volume" },
      num_companies: { $sum: 1 }
  }},
  { $sort: { avg_daily_volume: -1 } },
  { $project: {
      _id: 0,
      industry: "$_id",
      avg_daily_volume: { $round: ["$avg_daily_volume", 0] },
      num_companies: 1
  }}
])
